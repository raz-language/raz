  RAZ_RUNTIME_LIBRARY_PATH="$<TARGET_FILE:raz_runtime>")
if(TARGET oblink)
  target_compile_definitions(raz_forge_bridge PRIVATE RAZ_OBLINK_PATH="$<TARGET_FILE:oblink>")
  add_dependencies(raz_forge_bridge oblink)
endif()
# OpenSSL::SSL / OpenSSL::Crypto are imported targets, and find_package created
# them in src/runtime's directory scope, where imported targets stay by default.
# Testing for them here without finding them first is always false, which left
# the bridge -- and so every `raz build` -- unable to name the OpenSSL archives
# the runtime was compiled against. The stage-0 driver's block below finds them the same way.
find_package(OpenSSL QUIET)
if(OpenSSL_FOUND)
  target_compile_definitions(raz_forge_bridge PRIVATE
    RAZ_OPENSSL_SSL_LIBRARY_PATH="$<TARGET_LINKER_FILE:OpenSSL::SSL>"
    RAZ_OPENSSL_CRYPTO_LIBRARY_PATH="$<TARGET_LINKER_FILE:OpenSSL::Crypto>")
endif()
add_dependencies(raz_forge_bridge raz_runtime)

target_include_directories(raz_bootstrap_compiler PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(raz_bootstrap_compiler PUBLIC Forge::forge)
target_compile_features(raz_bootstrap_compiler PUBLIC cxx_std_20)

if(MSVC)
  target_compile_options(raz_bootstrap_compiler PRIVATE /W4 /permissive-)
  target_compile_options(raz_forge_bridge PRIVATE /W4 /permissive-)
  if(RAZ_ENABLE_WERROR)
    target_compile_options(raz_bootstrap_compiler PRIVATE /WX)
    target_compile_options(raz_forge_bridge PRIVATE /WX)
  endif()
else()
  target_compile_options(raz_bootstrap_compiler PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(raz_forge_bridge PRIVATE -Wall -Wextra -Wpedantic)
  # The native compiler is a temporary seed for the Raz-written production
  # compiler. Release-wide -O3 spends disproportionate time on the largest
  # compatibility translation units for little bootstrap benefit. O2 keeps the
  # seed fast at runtime, while the exceptionally large semantic analyzer uses
  # O1 to avoid pathological native optimizer time. Forge/runtime/production
  # compiler optimization is unaffected.
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_compile_options(raz_bootstrap_compiler PRIVATE -O2)
    set_source_files_properties(
      compiler/semantic/semantic_analyzer.cpp
      PROPERTIES COMPILE_OPTIONS "-O1")
  endif()
  if(RAZ_ENABLE_WERROR)
    target_compile_options(raz_bootstrap_compiler PRIVATE -Werror)
    target_compile_options(raz_forge_bridge PRIVATE -Werror)
  endif()
endif()


# Generate a static diagnostic-code catalog from the compiler sources.  The
# project driver exposes this to IDEs/CI through `raz diagnostics`, so the
# published code list cannot drift from codes actually present in the frontend.
file(GLOB_RECURSE RAZ_DIAGNOSTIC_CATALOG_SOURCES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/compiler/*.cpp"
  "${CMAKE_CURRENT_SOURCE_DIR}/compiler/*.hpp")
set(RAZ_DIAGNOSTIC_CODES "")
foreach(RAZ_DIAGNOSTIC_SOURCE IN LISTS RAZ_DIAGNOSTIC_CATALOG_SOURCES)
  file(READ "${RAZ_DIAGNOSTIC_SOURCE}" RAZ_DIAGNOSTIC_SOURCE_TEXT)
  string(REGEX MATCHALL "D[0-9][0-9][0-9][0-9]" RAZ_DIAGNOSTIC_SOURCE_CODES
    "${RAZ_DIAGNOSTIC_SOURCE_TEXT}")
  list(APPEND RAZ_DIAGNOSTIC_CODES ${RAZ_DIAGNOSTIC_SOURCE_CODES})
endforeach()
list(REMOVE_DUPLICATES RAZ_DIAGNOSTIC_CODES)
list(SORT RAZ_DIAGNOSTIC_CODES)
list(LENGTH RAZ_DIAGNOSTIC_CODES RAZ_DIAGNOSTIC_CODE_COUNT)
set(RAZ_DIAGNOSTIC_CATALOG_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${RAZ_DIAGNOSTIC_CATALOG_DIR}")
set(RAZ_DIAGNOSTIC_CATALOG_FILE "${RAZ_DIAGNOSTIC_CATALOG_DIR}/diagnostic_catalog.hpp")
set(RAZ_DIAGNOSTIC_CATALOG_TEMP "${RAZ_DIAGNOSTIC_CATALOG_DIR}/diagnostic_catalog.hpp.tmp")
file(WRITE "${RAZ_DIAGNOSTIC_CATALOG_TEMP}" "// Generated by CMake from compiler diagnostic literals.\n")
foreach(RAZ_DIAGNOSTIC_CODE IN LISTS RAZ_DIAGNOSTIC_CODES)
  file(APPEND "${RAZ_DIAGNOSTIC_CATALOG_TEMP}" "\"${RAZ_DIAGNOSTIC_CODE}\",\n")
endforeach()
# configure_file(COPYONLY) updates the destination only when contents differ.
# Avoid touching the stage-0 driver's generated include on every no-op configure.
configure_file("${RAZ_DIAGNOSTIC_CATALOG_TEMP}" "${RAZ_DIAGNOSTIC_CATALOG_FILE}" COPYONLY)
file(REMOVE "${RAZ_DIAGNOSTIC_CATALOG_TEMP}")

if(RAZ_BUILD_TOOLS)
  add_executable(razc_host tools/razc/main.cpp)
  target_link_libraries(razc_host PRIVATE Raz::bootstrap_compiler)
  set_target_properties(razc_host PROPERTIES OUTPUT_NAME "razc-stage0")

  add_executable(raz_host tools/raz/main.cpp)
  target_link_libraries(raz_host PRIVATE Raz::bootstrap_compiler)
  set_target_properties(raz_host PROPERTIES OUTPUT_NAME "raz-stage0")
  target_include_directories(raz_host PRIVATE "${RAZ_DIAGNOSTIC_CATALOG_DIR}")

  # The production compiler intentionally uses recursive syntax/HIR lowering.
  # Linux normally provides an ~8 MiB process stack, while PE/COFF defaults to
  # only 1 MiB.  Reserve the same 8 MiB on Windows so compiler-sized inputs do
  # not fail with STATUS_STACK_OVERFLOW (0xC00000FD).
  if(MSVC)
    target_link_options(razc_host PRIVATE /STACK:8388608)
    target_link_options(raz_host PRIVATE /STACK:8388608)
  endif()
  target_compile_definitions(raz_host PRIVATE
    RAZ_RUNTIME_LIBRARY_PATH="$<TARGET_FILE:raz_runtime>"
    RAZ_FORGE_BRIDGE_LIBRARY_PATH="$<TARGET_FILE:raz_forge_bridge>"
    RAZ_FORGE_LIBRARY_PATH="$<TARGET_FILE:forge>"
    RAZ_SOURCE_ROOT="${PROJECT_SOURCE_DIR}")
  if(TARGET oblink)
    target_compile_definitions(raz_host PRIVATE RAZ_OBLINK_PATH="$<TARGET_FILE:oblink>")
    add_dependencies(raz_host oblink)
  endif()
  find_package(OpenSSL QUIET)
  if(OpenSSL_FOUND)
    target_compile_definitions(raz_host PRIVATE
      RAZ_OPENSSL_SSL_LIBRARY_PATH="$<TARGET_LINKER_FILE:OpenSSL::SSL>"
      RAZ_OPENSSL_CRYPTO_LIBRARY_PATH="$<TARGET_LINKER_FILE:OpenSSL::Crypto>")
  endif()
  add_dependencies(raz_host raz_runtime raz_forge_bridge forge)
endif()
